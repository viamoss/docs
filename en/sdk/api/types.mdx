# TypeScript Types

The npm package ships TypeScript declarations. Import types from the package root:

```tsx
import type {
  MossSDKConfig,
  AgentState,
  ChatMessage,
  DefaultModalPosition,
  StabilityConfig,
} from '@viamoss/moss-sdk';
import { LogLevel } from '@viamoss/moss-sdk'; // an enum, so a value import
```

The types below match 0.17.0. Your editor shows the same documentation from the package's declarations.

---

## Core Types

### AgentState

What `useAgent()` returns. See [Hooks](/en/sdk/api/hooks) for how each member behaves.

```typescript
interface AgentState {
  // Connection and chat
  isConnected: boolean;
  isChatOpen: boolean;
  messages: ChatMessage[];
  sendMessage: (text: string, attachments?: AttachmentRef[]) => Promise<void>; // attachments: from 0.17.0
  openChat: (defaultPosition?: DefaultModalPosition) => void;
  closeChat: () => void;
  toggleChat: (defaultPosition?: DefaultModalPosition) => void;
  startNewChat: () => Promise<void>;
  activeChatSessionId: string | null;

  // Progress
  isWaitingForResponse: boolean;
  isRetrievalExpected: boolean;

  // Settings in effect
  useVision: boolean;
  debugMode: boolean;

  // Guides
  chatMode: 'chat' | 'guide';
  suggestsGuideMode: boolean;
  planPhase: 'offered' | 'executing' | null | undefined;
  continueGuide: () => Promise<void>;
  acceptGuideSuggestion: () => Promise<void>;
}
```

<Note>
    The declared `AgentState` has a few more members, for Moss's own debug tools and for features in development. They are not covered here and can change without notice. From 0.17.0, `displayMode` is gone.
</Note>

### ChatMessage

One message in the chat.

```typescript
interface ChatMessage {
  id: string;
  sender: 'user' | 'assistant';
  text: string;
  timestamp: number;
  userFeedback?: 'helpful' | 'unhelpful';
  sources?: MessageSource[];
  isHallucinated?: boolean;
  attachments?: ChatMessageAttachment[]; // from 0.17.0
  problem?: MessageProblem;              // from 0.17.0
  activity?: MessageActivity;            // from 0.17.0
}
```

| Field | Type | Meaning |
|-------|------|---------|
| `id` | `string` | Unique message ID |
| `sender` | `'user' \| 'assistant'` | Who sent it |
| `text` | `string` | The text, which may contain Markdown |
| `timestamp` | `number` | When it was added, in milliseconds since the Unix epoch |
| `userFeedback` | `'helpful' \| 'unhelpful'` | The user's thumbs up or down on an answer |
| `sources` | `MessageSource[]` | Help articles the answer cites, shown under it |
| `isHallucinated` | `boolean` | `true` when the answer pointed at an element Moss couldn't find on the page |
| `attachments` | `ChatMessageAttachment[]` | Files sent with a user message: `{ filename: string }` each |
| `problem` | `MessageProblem` | Set on an answer whose turn had a problem |
| `activity` | `MessageActivity` | The steps Moss took for the turn behind an answer, when the backend reported them (`streamActivity`) |

```typescript
interface MessageSource {
  title: string;
  url?: string | null;
  source?: string | null;
  readTimeMinutes?: number | null;
  linkScope?: 'host' | 'prefix' | null;   // from 0.17.0
  linkPrefix?: string | null;             // from 0.17.0
}
```

From 0.17.0, these types describe a message's problem and the steps behind it:

```typescript
interface MessageProblem {
  failed: boolean;                     // the turn gave no answer; the text is the failure
  notes: string[];                     // what went wrong, one line each
  fixBy: 'user' | 'admin' | 'moss';    // who can act on it
  action?: 'retry' | 'continue';       // the step offered while this is the last message
  reference?: string;                  // what to quote to support
}

interface MessageActivity {
  steps: MessageActivityStep[];
}

interface MessageActivityStep {
  id: string;
  kind: string;                        // 'retrieval', a tool's name, or 'tool'
  state: 'ok' | 'error';
  result?: { count?: number; titles?: string[]; status?: string };
}

interface ChatMessageAttachment {
  filename: string;
}
```

## Configuration Types

### MossSDKConfig

The `config` prop of `AgentProvider`, and the shape of `window.mossSettings` on the CDN script. See [Configuration](/en/sdk/configuration) for what each option does.

```typescript
// Abridged: options reserved for Moss's own tools are left out.
type MossSDKConfig = {
  // Required
  apiUrl: string;
  userId: string;

  // Authentication
  jwt?: string;
  getJwt?: () => Promise<string>;
  getBridgeCode?: () => Promise<string>;               // from 0.17.0

  // Page data and privacy
  pageContext?: 'full' | 'metadata' | 'none';
  redactAllInputs?: boolean;
  redactionSelectors?: string[];
  useVision?: boolean;
  screenshotMode?: ScreenshotMode;
  enableSessionRecording?: boolean;
  enableScreenHistoryRecording?: boolean;
  observeTargetSelector?: string;
  userMetadata?: Record<string, string | number | boolean | string[]>;
  collectionEnabled?: boolean;                         // from 0.17.0
  collectionControl?: MossCollectionControl;           // from 0.17.0
  pageSanitizationScript?: (document: Document) => void;

  // Behavior
  language?: 'en' | 'ko';
  sessionConfig?: {
    inactivityTimeout?: number;                        // minutes
    maxSessionAge?: number;                            // hours
    showNotification?: boolean;
    onReturnAfterInactivity?: 'new_chat' | 'ask';
  };
  stability?: StabilityConfig;
  closingChatEndsGuide?: boolean;
  enableProactiveGuides?: boolean;
  streamActivity?: boolean;                            // from 0.17.0
  replyLinks?: { allowedHosts?: string[] };            // from 0.17.0

  // Appearance
  appearance?: AppearanceInput;
  pageKey?: string;

  // Callbacks
  onSupportTicket?: (handoff: SupportTicketHandoff) => Promise<SupportTicketOutcome>;
  onEvent?: (event: MossEvent) => void;
  eventQueryText?: 'omit' | 'include' | ((text: string) => string);
  onError?: (error: MossError) => void;                // from 0.17.0

  // Diagnostics
  logLevel?: LogLevel;
  debugMode?: boolean;
} & (
  | { applicationId: string; applicationName?: string }
  | { applicationName: string; applicationId?: string }
);
```

At least one of `applicationId` and `applicationName` is required; use `applicationId`. From 0.17.0, `displayMode`, `headless` and `forcedActivationFlow` are gone, and TypeScript rejects them. The exported types `HeadlessConfig`, `ForcedActivationFlowConfig` and `ForcedActivationFlowDefinition` are gone too, so an import of them no longer compiles.

`sessionConfig` has no exported type of its own. Refer to it as `NonNullable<MossSDKConfig['sessionConfig']>`. See [Sessions](/en/sdk/api/session).

### ScreenshotMode

```typescript
type ScreenshotMode = 'fullpage' | 'viewport';
```

It has no effect on chat screenshots in 0.17.0; see [Page data](/en/sdk/page-data#vision-screenshots).

### LogLevel

Console verbosity. An enum: pass `LogLevel.DEBUG`, not the string `'DEBUG'`.

```typescript
enum LogLevel {
  DEBUG = 1,
  INFO = 2,    // default
  WARN = 3,
  ERROR = 4,
  SILENT = 5,  // no logging
}
```

```tsx
import { LogLevel } from '@viamoss/moss-sdk';

<AgentProvider config={{
  // ...
  logLevel: LogLevel.WARN,
}}>
```

## Stability Types

See [Stability Tuning](/en/sdk/advanced#stability-tuning) for defaults and behavior.

### StabilityConfig

```typescript
interface StabilityConfig {
  afterReload?: {
    enabled?: boolean;                  // default false
    pendingTTL?: number;                // ms, default 30000
  };
  maxTotalWaitMs?: number;              // default 15000
  networkTimeoutMs?: number;            // default 10000
  browserIdleTimeoutMs?: number;        // default 500
  layers?: StabilityLayersConfig;
  loadingIndicators?: LoadingIndicatorsLayerConfig;
  layoutShift?: LayoutShiftLayerConfig;
  domMutations?: DOMMutationsLayerConfig;
  frameReadiness?: FrameReadinessLayerConfig;
  resourceQuiet?: ResourceQuietLayerConfig;
  placeholderGuard?: PlaceholderGuardConfig;
}
```

### StabilityLayersConfig

```typescript
interface StabilityLayersConfig {
  networkIdle?: boolean;        // default true
  domMutations?: boolean;       // default true
  loadingIndicators?: boolean;  // default false
  frameReadiness?: boolean;     // default true
  resourceQuiet?: boolean;      // default true
  layoutShift?: boolean;        // default false
  browserIdle?: boolean;        // default true
  finalFrame?: boolean;         // default true
}
```

### Layer settings

```typescript
interface DOMMutationsLayerConfig {
  enabled?: boolean;                 // false turns the layer off
  quietPeriodMs?: number;            // default 800
  excludeSelectors?: string[];
  countQuietBeforeLayer?: boolean;   // default true; from 0.17.0
}

interface LoadingIndicatorsLayerConfig {
  enabled?: boolean;                 // false turns the layer off
  selectors?: string[];              // replaces the built-in list
  additionalSelectors?: string[];    // adds to the built-in list
  excludeSelectors?: string[];
  timeoutMs?: number;                // default 5000
}

interface LayoutShiftLayerConfig {
  enabled?: boolean;                 // false turns the layer off
  quietPeriodMs?: number;            // default 200
  ignoreThreshold?: number;          // default 0.01
}

interface FrameReadinessLayerConfig {
  enabled?: boolean;                 // false turns the layer off
  quietMs?: number;                  // default 200
  timeoutMs?: number;                // default 10000
}

interface ResourceQuietLayerConfig {
  enabled?: boolean;                 // false turns the layer off
  quietMs?: number;                  // default 500
  timeoutMs?: number;                // default 8000
}

interface PlaceholderGuardConfig {
  enabled?: boolean;                 // default true
  timeoutMs?: number;                // default 4000; not counted in maxTotalWaitMs
  selectors?: string[];
  additionalSelectors?: string[];
  excludeSelectors?: string[];
}
```

A layer object's `enabled: true` does not turn on a layer that is off by default; use `layers` for that.

## Appearance Types

See [Appearance](/en/sdk/appearance) for what each part controls.

```typescript
type AppearanceInput = 'blue' | 'purple' | DeepPartial<MossAppearance>;

type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P];
};
```

`MossAppearance` is the full theme. Its top-level keys are `name`, `colors`, `typography`, `spacing`, `radius`, `shadows`, `effects`, `inputBar`, `progressBar`, `modal`, and the optional `layout`, `chrome`, `zeroState` and `fallbackStates`. Pass only the values you want to change:

```tsx
appearance: 'purple'                              // a preset

appearance: { colors: { primary: '#FF5722' } }    // overrides on the default preset
```

## Usage Example

```tsx
import {
  AgentProvider,
  AssistantButton,
  useAgent,
  LogLevel,
} from '@viamoss/moss-sdk';
import type { MossSDKConfig, AgentState } from '@viamoss/moss-sdk';

const config: MossSDKConfig = {
  apiUrl: 'https://moss-api.viamoss.ai',
  applicationId: 'your-app-id',
  userId: 'user-123',
  getJwt: async () => fetchToken(),
  logLevel: LogLevel.INFO,
};

function App() {
  return (
    <AgentProvider config={config}>
      <AssistantButton />
      <ChatStatus />
    </AgentProvider>
  );
}

function ChatStatus() {
  const { messages, isChatOpen }: AgentState = useAgent();

  return (
    <div>
      <p>Chat is {isChatOpen ? 'open' : 'closed'}</p>
      <p>{messages.length} messages</p>
    </div>
  );
}
```
